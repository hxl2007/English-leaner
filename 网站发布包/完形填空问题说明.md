### 方案 2：临时隐藏占位文本（快速上线）

如果需要快速上线，可以先隐藏这些占位文本，只显示第一段真实内容：

修改 `网站/练习室.js` 的 `篇章内容` 函数（第46-49行）：

```javascript
function 篇章内容(p){
  const 表格=p.表格?`<div class="文章表格"><table>${p.表格.列?`<thead><tr>${p.表格.列.map(x=>`<th>${转义(x)}</th>`).join('')}</tr></thead>`:''}<tbody>${p.表格.行.map(row=>`<tr>${row.map(x=>`<td>${转义(x)}</td>`).join('')}</tr>`).join('')}</tbody></table></div>`:'';
  
  // 过滤掉占位文本段落
  const 有效段落 = p.段落.filter(s => 
    !s.includes('The next part of the story') && 
    !s.includes('tests the word that best completes')
  );
  
  return 有效段落.map((s,i)=>`${i===(p.表格位置??1)?表格:''}<p>${转义(s).replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>').replace(/\{\{(\d+)\}\}/g,'<strong class="完形空">($1) ____</strong>')}</p>`).join('')+(p.表格位置===有效段落.length?表格:'');
}
```

### 方案 3：使用 Python OCR 脚本批量处理

如果您的电脑安装了 Python 和 Tesseract OCR，可以使用以下脚本：

```python
# 需要安装：pip install pytesseract pillow
import pytesseract
from PIL import Image
import os
import json

image_dir = r"D:\专生本英语题目学习软件\专升本英语完型填空"
output_file = "完形填空提取文本.json"

results = []
for filename in sorted(os.listdir(image_dir)):
    if filename.endswith(('.jpg', '.png', '.jpeg')):
        img_path = os.path.join(image_dir, filename)
        img = Image.open(img_path)
        text = pytesseract.image_to_string(img, lang='eng')
        results.append({
            "图片": filename,
            "文本": text
        })

with open(output_file, 'w', encoding='utf-8') as f:
    json.dump(results, f, ensure_ascii=False, indent=2)

print(f"提取完成！结果保存在 {output_file}")
```
